"""Authorized recovery only; frozen study modules are never edited.

The integration loop is compiled directly from the checksum-locked recorder.
Only its starting index and already-saved boundary guard are changed. All
arithmetic, safety checks, diagnostics and checkpoint writes remain verbatim AST.
"""

def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))

def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def exclusive_json(path, data):
    with Path(path).open('x', encoding='utf-8') as stream:
        json.dump(data, stream, indent=2, allow_nan=False)

def verify_sources():
    inventory = read(F/'pilot-complete/source-inventory.json')
    for group, base in [('fresh_runtime', H.parent), ('fresh_implementation', F), ('predecessor', H)]:
        for rel, digest in inventory[group].items():
            if rh.sha(base/rel) != digest:
                raise ValueError('Frozen source changed: '+rel)
    if rh.sha(F/'protocol.json') != PROTOCOL_SHA:
        raise ValueError('Protocol changed')
    return inventory

def compiled_loop():
    if rh.sha(F/'record_history.py') != RECORDER_SHA:
        raise ValueError('Recorder source changed')
    tree = ast.parse((F/'record_history.py').read_text())
    function = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'run')
    loop = next(n for n in ast.walk(function) if isinstance(n, ast.For) and isinstance(n.target, ast.Name) and n.target.id == 'step')
    original = copy.deepcopy(loop)
    assert ast.unparse(loop.iter) == 'range(nstep + 1)'
    loop.iter = ast.parse('range(start_step, nstep + 1)', mode='eval').body
    sample_block = loop.body[2]
    assert isinstance(sample_block, ast.If) and ast.unparse(sample_block.test) == 'step % stride == 0'
    sample_block.test = ast.parse('step > start_step and step % stride == 0', mode='eval').body
    # Verify that reversing precisely these two scheduling edits restores the
    # entire original AST, including every arithmetic/safety/checkpoint node.
    restored = copy.deepcopy(loop)
    restored.iter = copy.deepcopy(original.iter)
    restored.body[2].test = copy.deepcopy(original.body[2].test)
    assert ast.dump(restored) == ast.dump(original)
    return compile(ast.fix_missing_locations(ast.Module(body=[loop], type_ignores=[])), str(F/'record_history.py')+' [checkpoint continuation]', 'exec')

def batch_with_preserved_logs(jobs, p, frozen, kind):
    """Frozen job/batch logic, routing new logs away from old empty launch logs."""
    verify_sources()
    tree=ast.parse((F/'controller.py').read_text())
    job=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='job')
    batch=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='batch')
    original=copy.deepcopy(job)
    opening=next(n for n in ast.walk(job) if isinstance(n,ast.Call) and ast.unparse(n)=="(HERE / (name + '.log')).open('w')")
    original_open=copy.deepcopy(opening)
    opening.func.value=ast.parse("RECOVERY_LOGS / (name + '.log')",mode='eval').body
    opening.args=[ast.Constant('x')]
    restored=copy.deepcopy(job)
    changed=next(n for n in ast.walk(restored) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and ast.unparse(n.func.value)=="RECOVERY_LOGS / (name + '.log')")
    changed.func=original_open.func;changed.args=original_open.args
    assert ast.dump(restored)==ast.dump(original)
    namespace=dict(ctl.__dict__);namespace['RECOVERY_LOGS']=OUT/'new-job-logs'
    namespace['RECOVERY_LOGS'].mkdir(exist_ok=True)
    exec(compile(ast.fix_missing_locations(ast.Module(body=[job,batch],type_ignores=[])),str(F/'controller.py')+' [separate logs]','exec'),namespace)
    return namespace['batch'](jobs,p,frozen,kind)

def load_checkpoint(checkpoint, expected, config, seed):
    checkpoint = Path(checkpoint)
    if rh.sha(checkpoint) != expected['sha256'] or rh.sha(checkpoint.with_suffix('.json')) != expected['metadata_sha256']:
        raise ValueError('Checkpoint identity mismatch')
    m = read(checkpoint.with_suffix('.json'))
    if m['checkpoint_sha256'] != expected['sha256'] or m['config'] != config or m['seed'] != seed or m['sources'] != rh.sources():
        raise ValueError('Checkpoint configuration/source mismatch')
    with np.load(checkpoint, allow_pickle=False) as z:
        z = {k: z[k].copy() for k in z.files}
    if not all(np.isfinite(v).all() for v in z.values()):
        raise ValueError('Nonfinite checkpoint')
    step = int(z['step']); stride = round(config['save_dt']/config['dt'])
    if step != expected['step'] or step % stride or step * config['dt'] != m['elapsed']:
        raise ValueError('Checkpoint time/index mismatch')
    count = step//stride + 1
    for key in KEYS:
        if len(z['saved_'+key]) != count:
            raise ValueError('Incomplete checkpoint samples')
    np.testing.assert_array_equal(z['saved_time'], np.arange(count)*stride*config['dt'])
    if len(m['diagnostics']) != count or m['diagnostics'][-1]['t'] != m['elapsed']:
        raise ValueError('Checkpoint diagnostics mismatch')
    np.testing.assert_array_equal(z['positions'], z['saved_positions'][-1])
    np.testing.assert_array_equal(z['tangent'], z['saved_tangent'][-1])
    np.testing.assert_array_equal(z['saved_positions'][0], np.random.default_rng(seed+80000).uniform(-rh.L/2, rh.L/2, (config['markers'],3)))
    np.testing.assert_array_equal(z['saved_tangent'][0], np.broadcast_to(np.eye(3),(config['markers'],3,3)))
    return m, z

def resume_checkpoint(seed, config, checkpoint, expected, destination, runtime):
    """No initialization or earlier time steps; no writes to original evidence."""
    verify_sources()
    m, z = load_checkpoint(checkpoint, expected, config, seed)
    name = rh.name_for(seed, config)
    destination = Path(destination); runtime = Path(runtime)
    destination.mkdir(parents=True, exist_ok=True); runtime.mkdir(parents=True, exist_ok=True)
    path = destination/(name+'.npz')
    if path.exists() or path.with_suffix('.json').exists():
        raise ValueError('Refusing to overwrite completed archive')
    claim = runtime/(name+'-claim.json')
    exclusive_json(claim, {'pid':os.getpid(), 'status':'running', 'seed':seed, 'start_step':int(z['step']), 'original_checkpoint':str(checkpoint), 'original_checkpoint_sha256':expected['sha256'], 'recovery_wrapper_sha256':rh.sha(Path(__file__)), 'started_UTC':now()})
    started = now(); clock_start = time.perf_counter()
    f = rh.Flow(config['n'], config['nu'], workers=1)
    state = dict(rh.__dict__)
    state.update(seed=seed, config=config, name=name, f=f, h=z['field'], x=z['positions'], M=z['tangent'],
        start_step=int(z['step']), dt=config['dt'], nstep=round(config['end']/config['dt']), stride=round(config['save_dt']/config['dt']),
        arrays={k:list(z['saved_'+k]) for k in KEYS}, diag=copy.deepcopy(m['diagnostics']), energy0=m['diagnostics'][0]['energy'],
        integrated_diss=z['integrated_diss'].item(), max_cfl=z['max_cfl'].item(), frozen=m['sources'], checkpoint=runtime/(name+'_latest.npz'))
    try:
        exec(compiled_loop(), state, state)
        verify_sources()
        # Stored prefix must survive unchanged; exactly one sample per time.
        for key in KEYS:
            np.testing.assert_array_equal(np.asarray(state['arrays'][key])[:len(z['saved_'+key])], z['saved_'+key])
        assert state['diag'][:len(m['diagnostics'])] == m['diagnostics']
        if rh.sha(checkpoint) != expected['sha256'] or rh.sha(Path(checkpoint).with_suffix('.json')) != expected['metadata_sha256']:
            raise ValueError('Original checkpoint changed during continuation')
        # Opening exclusively prevents a racing process from being overwritten.
        with path.open('xb') as stream:
            np.savez_compressed(stream, **state['arrays'])
        meta = {'name':name, 'independent_group':seed, 'seed':seed, 'config':config,
            'sources':{p.name:rh.sha(p) for p in [H/'simulate.py',H.parent/'numerics.py',H/'protocol.json']},
            'study_sources':m['sources'], 'sha256':rh.sha(path), 'seconds':time.perf_counter()-clock_start,
            'max_cfl':state['max_cfl'], 'diagnostics':state['diag'], 'Re_Urms_L_over_nu':rh.L/config['nu'],
            'started_UTC':started, 'completed_UTC':now(),
            'checkpoint_continuation':{'original_checkpoint_sha256':expected['sha256'], 'original_checkpoint_metadata_sha256':expected['metadata_sha256'],
                'start_step':int(z['step']), 'final_step':state['nstep'], 'completed_updates':state['nstep']-int(z['step']),
                'saved_prefix_bitwise_preserved':True, 'original_diagnostics_preserved':True, 'seconds_scope':'continuation only; original elapsed wall time unavailable',
                'wrapper_sha256':rh.sha(Path(__file__)), 'frozen_recorder_sha256':RECORDER_SHA}}
        exclusive_json(path.with_suffix('.json'), meta)
        rh.validate(path, config)
        rh.save_json(claim, {**read(claim), 'status':'complete', 'sha256':meta['sha256'], 'completed_UTC':now()})
        print(name+' resumed and verified', flush=True)
        return path
    except Exception as exc:
        rh.save_json(claim, {**read(claim), 'status':'failed', 'error':repr(exc), 'failed_UTC':now()})
        raise

def inventory_preserved():
    files = [F/'controller.json', F/'execution.json']
    files += list((F/'claims').glob('*.json')) + list(F.glob('*.log'))
    files += list((F/'recovery').glob('*'))
    files += list((F/'recorded-data').glob('*'))
    return {str(path.relative_to(ROOT)).replace('\\','/'):rh.sha(path) for path in files if path.is_file()}

    p = read(F/'protocol.json'); ctl.review_gate(p)
    review = read(ROOT/'outputs/fresh-20261013-pilot-review.json')
    publication = read(ROOT/'outputs/fresh-complete-pilot-publication.json')
    assert publication['checks_success']
    audit = read(F/'pilot-audit-112.json')
    assert audit['all_screens_pass'] and len(audit['screens']) == 116
    assert rh.sha(F/'pilot-audit-112.json') == '6d9e1788fc4b4318d1c88393d9a6461211e5e0aabcf9e80e8f8130be7eb9e129'
    paths = sorted((F/'recorded-data').glob('*.npz'))
    assert len(paths) == 28, len(paths)
    configs = {rh.name_for(s,c):c for s,c in ctl.pilot_jobs([56,80,112,160],p)}
    configs.update({rh.name_for(s,{**p['base'],'n':112}):{**p['base'],'n':112} for s in p['train_seeds'][:6]})
    assert set(x.stem for x in paths) == set(configs)
    for path in paths:
        rh.validate(path, configs[path.stem])
    checkpoints = {x['seed']:x for x in review['preserved_partial_checkpoints']}
    assert sorted(checkpoints) == PARTIAL
    c = {**p['base'],'n':112}
    for seed in PARTIAL:
        assert checkpoints[seed]['step'] == 440
        load_checkpoint(F/'recovery'/(rh.name_for(seed,c)+'_latest.npz'), checkpoints[seed], c, seed)
    pending = [(s,c) for s in p['train_seeds'][9:]+p['validation_seeds']]
    assert len(pending) == 39
    for seed,c in pending:
        name = rh.name_for(seed,c)
        for path in [F/'recorded-data'/(name+'.npz'), F/'recorded-data'/(name+'.json'), F/'claims'/(name+'.json'), F/'recovery'/(name+'_latest.npz')]:
            assert not path.exists(), str(path)
        old_log=F/(name+'.log')
        if old_log.exists():assert old_log.stat().st_size==0, str(old_log)
    assert not list(F.glob('models-*/*.joblib')) and not (F/'test-lock.json').exists() and not (F/'numerical-stop.json').exists()
    compiled_loop()
    return p, checkpoints, pending

def self_test():
    """Tiny synthetic recorder tests outside study data, never study trajectories."""
    base = OUT/('self-test-'+datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%S')); base.mkdir(parents=True, exist_ok=False)
    results = []
    for seed,dt in [(137,.0025),(138,.00125)]:
        case = base/str(seed); captured = case/'captured'; captured.mkdir(parents=True)
        c = {'n':8,'nu':.02,'dt':dt,'save_dt':.025,'end':.2,'markers':16}
        original_save = rh.save_json
        def capture(path,obj):
            original_save(path,obj)
            if 'checkpoint_sha256' in obj:
                label = 'boundary' if obj['elapsed'] == .1 else 'final'
                src=Path(path).with_suffix('.npz')
                shutil.copyfile(src, captured/(label+'.npz'))
                shutil.copyfile(path, captured/(label+'.json'))
        rh.save_json = capture
        try:
            original = rh.run(seed,c,case/'reference/recorded-data')
        finally:
            rh.save_json = original_save
        checkpoint = captured/'boundary.npz'
        expected = {'sha256':rh.sha(checkpoint), 'metadata_sha256':rh.sha(checkpoint.with_suffix('.json')), 'step':round(.1/dt)}
        result = resume_checkpoint(seed,c,checkpoint,expected,case/'resumed',case/'runtime')
        with np.load(original) as a, np.load(result) as b:
            for key in a.files:
                np.testing.assert_array_equal(a[key],b[key])
        final = case/'runtime'/(rh.name_for(seed,c)+'_latest.npz')
        with np.load(captured/'final.npz') as a, np.load(final) as b:
            for key in a.files:
                np.testing.assert_array_equal(a[key],b[key])
        ma=read(original.with_suffix('.json'));mb=read(result.with_suffix('.json'))
        assert ma['diagnostics']==mb['diagnostics'] and ma['max_cfl']==mb['max_cfl']
        rejections=[]
        for label,action in [
            ('changed_hash',lambda:load_checkpoint(checkpoint,{**expected,'sha256':'0'*64},c,seed)),
            ('changed_config',lambda:load_checkpoint(checkpoint,expected,{**c,'nu':.04},seed)),
            ('changed_step',lambda:load_checkpoint(checkpoint,{**expected,'step':expected['step']+1},c,seed)),
            ('overwrite',lambda:resume_checkpoint(seed,c,checkpoint,expected,case/'resumed',case/'runtime')),
            ('duplicate_owner',lambda:resume_checkpoint(seed,c,checkpoint,expected,case/'duplicate',case/'runtime'))]:
            try: action()
            except (ValueError,FileExistsError): rejections.append(label)
            else: raise AssertionError('Failed rejection '+label)
        results.append({'synthetic_seed':seed,'grid':8,'dt':dt,'archive_arrays_exact':True,'final_field_and_state_exact':True,'all_diagnostics_and_CFL_exact':True,'rejections':rejections})
    execution_before=rh.sha(F/'execution.json')
    batch_with_preserved_logs([],read(F/'protocol.json'),rh.sources(),'software-test-empty')
    assert rh.sha(F/'execution.json')==execution_before
    rh.save_json(OUT/'self-test.json', {'passed':True,'at_UTC':now(),'wrapper_sha256':rh.sha(Path(__file__)),'tests':results,'separate_log_batch_AST_verified':True,'research_trajectories_generated':0})
    print(json.dumps({'self_test_passed':True,'cases':len(results),'all_arrays_and_final_fields_bitwise_equal':True}),flush=True)

def run_preparation():
    authorization = read(OUT/'authorization.json')
    assert authorization['authorized'] and authorization['user_message'] == 'go'
    proof = read(OUT/'self-test.json')
    assert proof['passed'] and proof['wrapper_sha256'] == rh.sha(Path(__file__))
    ownership = read(OUT/'ownership-before-launch.json')
    assert ownership['original_processes_absent'] and ownership['fresh_scope_processes_absent']
    age = datetime.datetime.now(datetime.timezone.utc)-datetime.datetime.fromisoformat(ownership['checked_UTC'])
    assert age.total_seconds() < 180, 'Ownership inventory too old'
    p, checkpoints, pending = preflight()
    baseline = inventory_preserved()
    exclusive_json(OUT/'preserved-before.json',baseline)
    # Durable second copy of all stale records (bulk checkpoints remain in place,
    # checksum protected and never overwritten by the continuation).
    for rel in baseline:
        src=ROOT/rel
        if src.suffix != '.npz':
            dest=OUT/'original-records'/src.relative_to(F)
            dest.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(src,dest)
    lock=OUT/'controller.json'
    exclusive_json(lock,{'pid':os.getpid(),'status':'running','phase':'checkpoint-continuation','started_UTC':now(),'sources':rh.sources(),'wrapper_sha256':rh.sha(Path(__file__))})
    frozen=rh.sources();c={**p['base'],'n':112}
    try:
        with ThreadPoolExecutor(max_workers=p['workers']) as pool:
            jobs=[]
            for seed in PARTIAL:
                def work(s=seed):
                    with (OUT/(str(s)+'.log')).open('x') as log:
                        child=subprocess.Popen([sys.executable,'-B','-u',str(Path(__file__)),'--resume-seed',str(s)],stdout=log,stderr=subprocess.STDOUT,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
                        rh.save_json(OUT/(str(s)+'-launcher.json'),{'pid':child.pid,'parent':os.getpid(),'seed':s,'started_UTC':now()})
                        if child.wait():raise RuntimeError('Checkpoint continuation failed: '+str(s))
                    return s
                jobs.append(pool.submit(work))
            for future in as_completed(jobs):
                print('Recovered seed '+str(future.result()),flush=True)
        # Validate and register all prior complete runs without rerunning them.
        ctl.batch([(s,c) for s in p['train_seeds'][:9]],p,frozen,'train-validation-112')
        rh.save_json(lock,{**read(lock),'phase':'train-validation','recovered':PARTIAL,'unstarted_jobs':39})
        batch_with_preserved_logs(pending,p,frozen,'train-validation-112')
        verify_sources();ctl.assert_test_absent(p)
        rh.save_json(lock,{**read(lock),'phase':'fitting'})
        ctl.script('fit_models.py','--grid',112)
        if not ctl.audit(112,'model'):
            ctl.stop('Candidate80 failed its pilot; fresh fitted predictors at candidate112 fail the predefined numerical screens')
            outcome='numerical-stop'
        else:
            ctl.assert_test_absent(p)
            manifest=F/'models-112/manifest.json'
            test_lock={'grid':112,'manifest':'models-112/manifest.json','manifest_sha256':rh.sha(manifest),'protocol_sha256':rh.sha(F/'protocol.json'),'sources':frozen,'locked_at_UTC':now(),'test_absent_at_lock':True,'model_numeric_audit_sha256':rh.sha(F/'model-audit-112.json')}
            exclusive_json(F/'test-lock.json',test_lock)
            rh.save_json(F/'waiting-model-publication.json',{'status':'Publish and independently verify models/lock before test phase','test_lock_sha256':rh.sha(F/'test-lock.json')})
            outcome='waiting-model-publication'
        # Only execution.json may change among the original preserved identities.
        for rel,digest in baseline.items():
            if Path(rel).name != 'execution.json':assert rh.sha(ROOT/rel)==digest,rel
        rh.save_json(lock,{**read(lock),'status':'complete','phase':outcome,'completed_UTC':now()})
        print(outcome+'; no untouched tests launched',flush=True)
    except Exception as exc:
        rh.save_json(lock,{**read(lock),'status':'failed','error':repr(exc),'failed_UTC':now()})
        raise

def main():
    ap=argparse.ArgumentParser();ap.add_argument('--self-test',action='store_true');ap.add_argument('--preflight',action='store_true');ap.add_argument('--resume-seed',type=int);ap.add_argument('--prepare',action='store_true');a=ap.parse_args()
    OUT.mkdir(parents=True,exist_ok=True)
    if a.self_test:self_test()
    elif a.preflight:
        p,cp,pending=preflight();print(json.dumps({'preflight_passed':True,'preserved_complete':28,'partial':list(cp),'unstarted':len(pending)}))
    elif a.resume_seed is not None:
        assert a.resume_seed in PARTIAL
        lock=read(OUT/'controller.json');assert lock['status']=='running' and lock['phase']=='checkpoint-continuation'
        assert lock['wrapper_sha256']==rh.sha(Path(__file__))
        p=read(F/'protocol.json');c={**p['base'],'n':112}
        expected=next(x for x in read(ROOT/'outputs/fresh-20261013-pilot-review.json')['preserved_partial_checkpoints'] if x['seed']==a.resume_seed)
        resume_checkpoint(a.resume_seed,c,F/'recovery'/(rh.name_for(a.resume_seed,c)+'_latest.npz'),expected,F/'recorded-data',OUT/'continued-checkpoints')
    elif a.prepare:run_preparation()
    else:ap.error('Choose one operation')

if __name__=='__main__':main()
